SQL DDL SCHEMA & RLS POLICIES (PRODUCTION)

Salin dan jalankan script SQL ini pada Supabase Dashboard → SQL Editor untuk membuat seluruh tabel, hak akses RLS, dan storage bucket publik. 

-- ==========================================================
-- TOOLTRACK PRO ENTERPRISE - SQL DATABASE SCHEMA & POLICIES
-- ==========================================================

-- 1. TABEL USERS (Petugas & Superadmin)
CREATE TABLE IF NOT EXISTS public.users (
  id BIGSERIAL PRIMARY KEY,
  nik VARCHAR(50) UNIQUE NOT NULL,
  name VARCHAR(150) NOT NULL,
  password VARCHAR(100) NOT NULL,
  role VARCHAR(50) DEFAULT 'petugas',
  office VARCHAR(100) DEFAULT 'Workshop Utama',
  avatar TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Seed Default Accounts
INSERT INTO public.users (nik, name, password, role, office)
VALUES 
  ('ADMIN01', 'Super Administrator', 'admin123', 'superadmin', 'Workshop Utama'),
  ('TECH01', 'Teknisi Lapangan', 'tech123', 'petugas', 'Workshop Utama')
ON CONFLICT (nik) DO NOTHING;

-- 2. TABEL TOOLS (Katalog Peralatan & Stok)
CREATE TABLE IF NOT EXISTS public.tools (
  id VARCHAR(50) PRIMARY KEY,
  name VARCHAR(255) NOT NULL,
  category VARCHAR(100) NOT NULL,
  brand VARCHAR(100) NOT NULL,
  office VARCHAR(100) NOT NULL,
  total_qty INTEGER DEFAULT 1,
  available_qty INTEGER DEFAULT 1,
  status VARCHAR(50) DEFAULT 'TERSEDIA',
  photo TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 3. TABEL LOANS (Transaksi Peminjaman & Audit)
CREATE TABLE IF NOT EXISTS public.loans (
  id VARCHAR(50) PRIMARY KEY,
  tool_id VARCHAR(50) REFERENCES public.tools(id) ON DELETE CASCADE,
  tool_name VARCHAR(255) NOT NULL,
  qty_borrowed INTEGER DEFAULT 1,
  borrower_name VARCHAR(150) NOT NULL,
  borrower_phone VARCHAR(50) NOT NULL,
  officer_borrow VARCHAR(150) NOT NULL,
  borrow_date VARCHAR(50) NOT NULL,
  notes_borrow TEXT,
  photo_before TEXT,
  before_size_kb INTEGER,
  rent_started_at BIGINT,
  return_date VARCHAR(50),
  officer_return VARCHAR(150),
  condition_return VARCHAR(100),
  notes_return TEXT,
  photo_after TEXT,
  after_size_kb INTEGER,
  status VARCHAR(50) DEFAULT 'DIPINJAM',
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 4. TABEL OFFICES (Master Workshop & Gudang)
CREATE TABLE IF NOT EXISTS public.offices (
  id BIGSERIAL PRIMARY KEY,
  name VARCHAR(150) UNIQUE NOT NULL,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

INSERT INTO public.offices (name)
VALUES ('Workshop Utama'), ('Gudang Proyek Barat'), ('Gudang Proyek Timur')
ON CONFLICT (name) DO NOTHING;

-- 5. ROW LEVEL SECURITY (RLS) FULL ACCESS POLICIES
ALTER TABLE public.users ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Public Full Users" ON public.users;
CREATE POLICY "Public Full Users" ON public.users FOR ALL TO anon, authenticated USING (true) WITH CHECK (true);

ALTER TABLE public.tools ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Public Full Tools" ON public.tools;
CREATE POLICY "Public Full Tools" ON public.tools FOR ALL TO anon, authenticated USING (true) WITH CHECK (true);

ALTER TABLE public.loans ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Public Full Loans" ON public.loans;
CREATE POLICY "Public Full Loans" ON public.loans FOR ALL TO anon, authenticated USING (true) WITH CHECK (true);

ALTER TABLE public.offices ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Public Full Offices" ON public.offices;
CREATE POLICY "Public Full Offices" ON public.offices FOR ALL TO anon, authenticated USING (true) WITH CHECK (true);

-- 6. STORAGE BUCKET CONFIGURATION (Foto <= 100KB)
INSERT INTO storage.buckets (id, name, public) 
VALUES ('tool_photos', 'tool_photos', true)
ON CONFLICT (id) DO NOTHING;

DROP POLICY IF EXISTS "Public Storage Access" ON storage.objects;
CREATE POLICY "Public Storage Access" ON storage.objects
FOR ALL TO anon, authenticated
USING (bucket_id = 'tool_photos')
WITH CHECK (bucket_id = 'tool_photos');
