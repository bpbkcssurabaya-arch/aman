Script Skema & Izin RLS Supabase

Jika Anda baru membuat tabel atau tombol aksi tidak merespons, jalankan script ini di menu SQL Editor di dashboard Supabase Anda:

-- 1. Buka Akses Penuh Tabel LOANS (Pinjam & Kembali)
ALTER TABLE public.loans ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Public Full Access Loans" ON public.loans;
CREATE POLICY "Public Full Access Loans" ON public.loans FOR ALL TO anon, authenticated USING (true) WITH CHECK (true);

-- 2. Buka Akses Penuh Tabel TOOLS (Katalog & Stok)
ALTER TABLE public.tools ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Public Full Access Tools" ON public.tools;
CREATE POLICY "Public Full Access Tools" ON public.tools FOR ALL TO anon, authenticated USING (true) WITH CHECK (true);

-- 3. Buka Akses Penuh Tabel USERS & OFFICES
ALTER TABLE public.users ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Public Full Access Users" ON public.users;
CREATE POLICY "Public Full Access Users" ON public.users FOR ALL TO anon, authenticated USING (true) WITH CHECK (true);

ALTER TABLE public.offices ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Public Full Access Offices" ON public.offices;
CREATE POLICY "Public Full Access Offices" ON public.offices FOR ALL TO anon, authenticated USING (true) WITH CHECK (true);

-- 4. Pastikan Kolom Pengembalian Ada
ALTER TABLE public.loans ADD COLUMN IF NOT EXISTS return_date TEXT;
ALTER TABLE public.loans ADD COLUMN IF NOT EXISTS officer_return TEXT;
ALTER TABLE public.loans ADD COLUMN IF NOT EXISTS condition_return TEXT;
ALTER TABLE public.loans ADD COLUMN IF NOT EXISTS notes_return TEXT;
ALTER TABLE public.loans ADD COLUMN IF NOT EXISTS photo_after TEXT;
ALTER TABLE public.loans ADD COLUMN IF NOT EXISTS after_size_kb INTEGER;
