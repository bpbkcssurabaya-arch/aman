async function uploadToSupabaseStorage(base64Data, filename) { if (!supabaseClient || !base64Data || !base64Data.startsWith('data:image')) { return base64Data; } try { const res = await fetch(base64Data); const blob = await res.blob(); const filePath = `uploads/${Date.now()}_${filename}.jpg`; // 1. Upload berkas foto fisik (≤ 100 KB) ke Bucket "loan_photos" const { data, error } = await supabaseClient.storage .from('loan_photos') .upload(filePath, blob, { contentType: 'image/jpeg', upsert: true }); if (error) { console.warn("Storage upload error:", error.message); return base64Data; // fallback jika bucket belum dibuat } // 2. Mengambil Public URL dari Supabase Storage const { data: publicData } = supabaseClient.storage .from('loan_photos') .getPublicUrl(filePath); return publicData.publicUrl; // URL ini yang disimpan ke database PostgreSQL } catch (e) { return base64Data; } }
